WITH filtered AS (
    SELECT 
        a.username,
        SUBSTRING_INDEX(d.name, '.', -1) AS tld,
        CAST(p.price AS DECIMAL(10,2)) AS price
    FROM domains d
    JOIN accounts a
        ON d.account_id = a.id
    JOIN prices p
        ON SUBSTRING_INDEX(d.name, '.', -1) = p.tld
    WHERE DATE_FORMAT(
        STR_TO_DATE(d.expiration_date, '%Y-%m-%d'),
        '%Y-%m'
    ) = '2022-08'
),

aggregated AS (
    SELECT
        username,
        tld,
        price,
        COUNT(*) AS domain_count,
        ROUND(COUNT(*) * price, 2) AS total_cost
    FROM filtered
    GROUP BY username, tld, price
)

SELECT
    username,

    MAX(
        CASE 
            WHEN tld = 'com'
            THEN CONCAT('$', FORMAT(total_cost, 2))
        END
    ) AS `.com ($10.00/year) x 1`,

    MAX(
        CASE 
            WHEN tld = 'org'
            THEN CONCAT('$', FORMAT(total_cost, 2))
        END
    ) AS `.org ($8.00/year) x 1`,

    MAX(
        CASE 
            WHEN tld = 'net'
            THEN CONCAT('$', FORMAT(total_cost, 2))
        END
    ) AS `.net ($5.00/year) x 1`

FROM aggregated
GROUP BY username
ORDER BY username;
